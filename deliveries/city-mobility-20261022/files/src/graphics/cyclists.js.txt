import * as T from '../../vendor/three.module.js';
import {addAssetInstances} from './assets.js';
import {prepareTraffic,advanceTraffic} from './traffic.js';
import {seedRoadUsers} from './road-network.js';
import {personLook} from './people.js';

export function cycleLeg(side,angle){
  const dy=side*(.15*Math.cos(angle)-.08*Math.sin(angle)),dz=side*(.15*Math.sin(angle)+.08*Math.cos(angle));
  const hip=new T.Vector3(side*.10,1.05,-.22),ankle=new T.Vector3(side*.11,.47+dy,-.06+dz);
  const y=ankle.y-hip.y,z=ankle.z-hip.z,d=Math.hypot(y,z),a=(.44**2-.46**2+d*d)/(2*d),b=Math.sqrt(Math.max(0,.44**2-a*a));
  const knee=new T.Vector3(side*.105,hip.y+a*y/d+b*z/d,hip.z+a*z/d-b*y/d);
  return {hip,knee,ankle,pedal:new T.Vector3(side*.13,.35+dy,-.02+dz)};
}

function boneTransform(a,b,c,d,result){
  const before=b.clone().sub(a),after=d.clone().sub(c),q=new T.Quaternion().setFromUnitVectors(before.normalize(),after.normalize());
  const midpoint=a.clone().add(b).multiplyScalar(.5),target=c.clone().add(d).multiplyScalar(.5);
  result.makeTranslation(...target.toArray()).multiply(new T.Matrix4().makeRotationFromQuaternion(q)).multiply(new T.Matrix4().makeTranslation(-midpoint.x,-midpoint.y,-midpoint.z));
}

export function buildCyclists(city){
  const users=seedRoadUsers(city.network.cyclePaths,[2,2,1,1,1,1,1,1],i=>({
    ...personLook(i+3),wheelRadius:.35,wheelbase:1.18,clearance:2.5,standOff:1.25,headway:1.0,maxAcceleration:1.0,cornerAcceleration:1.1,
    parts:{},initial:[cycleLeg(-1,0),cycleLeg(1,0)]
  }));
  const controller=city.cycleTraffic={cars:users,paths:city.network.cyclePaths,network:city.network};
  city.cyclists=users;prepareTraffic(controller);
  const batch=city.cyclistsBatch=addAssetInstances(city.root,'site-cyclist',users.map(()=>({position:[0,-.18,0]})),{lighting:true});
  city.batches.push(batch);
}

export function updateCyclists(city,dt,signals,reduced,wet,storm){
  const controller=city.cycleTraffic;
  if(!reduced)advanceTraffic(controller,dt,storm?1.2:wet?2.5:4.2,(axis,car,distance)=>signals[axis]==='green'||(signals[axis]==='amber'&&distance<1.25+car.speed*car.speed/5.2));
  const poses=city.cyclists.map(rider=>{
    const p=controller.paths[rider.lane].getPointAt(rider.phase);
    // Crank cadence is mechanically linked to travelled wheel distance; stopping
    // at a signal stops both the wheels and the rider's feet.
    const angle=rider.wheelAngle/2.65,parts=rider.parts;
    for(const [index,side] of [[0,-1],[1,1]]){
      const initial=rider.initial[index],current=cycleLeg(side,angle),suffix=side<0?'L':'R';
      for(const [name,a,b,c,d] of [[`CycleThigh${suffix}`,initial.hip,initial.knee,current.hip,current.knee],[`CycleCalf${suffix}`,initial.knee,initial.ankle,current.knee,current.ankle]]){
        parts[name] ||= new T.Matrix4();boneTransform(a,b,c,d,parts[name]);
      }
      for(const name of [`CycleShoe${suffix}`,`CyclePedal${suffix}`]){
        parts[name] ||= new T.Matrix4();parts[name].makeTranslation(0,current.pedal.y-initial.pedal.y,current.pedal.z-initial.pedal.z);
      }
    }
    return {...rider,position:[p.x,-.18,p.z],rotation:rider.heading,wheelAngle:rider.wheelAngle,crankAngle:angle,parts};
  });
  city.cyclistsBatch.setPoses(poses);
}
